import { memo, useMemo } from 'react';
import { Linking, View } from 'react-native';
import { EnrichedMarkdownText, type LinkPressEvent } from 'react-native-enriched-markdown';
import { Lexer } from 'marked';
import remend from 'remend';

import { useDesignTheme } from './design/theme';
import { createMarkdownBlocks, textMarkdownSupported } from './markdown-blocks';
import { MarkdownText } from './markdown-text';
import { space } from './design/tokens';
import { markdownStyle } from './markdown-style';

const flags = { latexMath: false } as const;
const openLink = ({ url }: LinkPressEvent) => void Linking.openURL(url).catch(() => undefined);

const MarkdownBlockView = memo(function MarkdownBlockView({
  source,
  color,
  selectable = true,
  streaming = false,
  trailing = false,
}: {
  source: string;
  color?: string;
  selectable?: boolean;
  streaming?: boolean;
  trailing?: boolean;
}) {
  const theme = useDesignTheme();
  const ink = color ?? theme.color.fg;
  const style = useMemo(() => markdownStyle(theme, ink), [theme, ink]);
  const markdown = useMemo(() => (streaming ? remend(source) : source), [source, streaming]);
  const tokens = useMemo(() => Lexer.lex(markdown, { gfm: true }), [markdown]);
  if (
    !selectable &&
    Object.keys(tokens.links).length === 0 &&
    tokens.every((token) => !['code', 'table'].includes(token.type)) &&
    textMarkdownSupported(tokens)
  )
    return (
      <View style={{ marginBottom: trailing ? space.x3 : 0 }}>
        <MarkdownText tokens={tokens} color={ink} selectable={false} />
      </View>
    );
  return (
    <EnrichedMarkdownText
      markdown={markdown}
      markdownStyle={style}
      flavor="github"
      allowTrailingMargin={trailing}
      selectable={selectable}
      md4cFlags={flags}
      enableTaskListItemToggle={false}
      onLinkPress={openLink}
    />
  );
});

export const MarkdownView = memo(function MarkdownView({
  source,
  color,
  selectable = true,
  streaming = false,
}: {
  source: string;
  color?: string;
  selectable?: boolean;
  streaming?: boolean;
}) {
  const theme = useDesignTheme();
  const ink = color ?? theme.color.fg;
  const parse = useMemo(() => createMarkdownBlocks(), []);
  const blocks = useMemo(() => (selectable ? [] : parse(source)), [parse, selectable, source]);
  if (selectable)
    return <MarkdownBlockView source={source} color={ink} selectable streaming={streaming} />;
  return (
    <View>
      {blocks.map((block, index) => (
        <MarkdownBlockView
          key={block.key}
          source={block.source}
          color={ink}
          selectable={false}
          trailing={index < blocks.length - 1}
          streaming={streaming && index === blocks.length - 1}
        />
      ))}
    </View>
  );
});
